<script>
(function () {
  "use strict";

  var CONFIG = __XAPI_CONFIG__;

  function log() {
    if (CONFIG.debug && window.console && console.log) {
      console.log.apply(console, ["[xAPI]"].concat([].slice.call(arguments)));
    }
  }

  // ---------- helpers ----------
  function hostOf(u) {
    var m = /^[a-z][a-z0-9+.-]*:\/\/([^\/:?#]+)/i.exec(u || "");
    return m ? m[1] : null;
  }
  function getParam(str, key) {
    var m = new RegExp("[?&]" + key + "=([^&]+)").exec(str || "");
    return m ? decodeURIComponent(m[1].replace(/\+/g, " ")) : null;
  }
  // Every reachable frame URL, innermost first (also catches hash-routed params).
  function frameUrls() {
    var urls = [], w = window, g = 0;
    while (w && g++ < 12) {
      try { if (w.location && w.location.href) { urls.push(w.location.href); } } catch (e) {}
      if (w.parent === w) { break; }
      try { w = w.parent; } catch (e) { break; }
    }
    try { if (document.referrer) { urls.push(document.referrer); } } catch (e) {}
    return urls;
  }

  // ---------- domain gate ----------
  function detectHost() {
    var h = null;
    try {
      var ao = window.location.ancestorOrigins;
      if (ao && ao.length) { h = hostOf(ao[ao.length - 1]); }
    } catch (e) {}
    if (!h) { try { h = window.top.location.hostname || null; } catch (e) {} }
    if (!h) { try { h = window.parent.location.hostname || null; } catch (e) {} }
    if (!h) { try { h = hostOf(document.referrer); } catch (e) {} }
    if (!h) { try { h = window.location.hostname || null; } catch (e) {} }
    return h && h !== "null" ? h : null;
  }
  function domainAllowed() {
    var list = CONFIG.allowedDomains || [];
    if (!list.length) { return true; }
    var host = detectHost();
    if (!host) {
      log("Host could not be determined." + (CONFIG.strictDomain ? " Strict - skipping." : " Sending anyway."));
      return !CONFIG.strictDomain;
    }
    host = String(host).toLowerCase();
    for (var i = 0; i < list.length; i++) {
      var pat = String(list[i]).toLowerCase();
      if (pat.indexOf("*.") === 0) {
        var base = pat.slice(2);
        if (host === base || host.slice(-(base.length + 1)) === "." + base) { return true; }
      } else if (host === pat) { return true; }
    }
    log("Host '" + host + "' not in allowed list - skipping.");
    return false;
  }

  if (!domainAllowed()) { return; }

  if (CONFIG.onceKey) {
    try {
      if (window.localStorage && localStorage.getItem(CONFIG.onceKey)) {
        log("Already sent for this activity - skipping.");
        return;
      }
    } catch (e) {}
  }

  // ---------- SCORM API discovery ----------
  // Walks the frame chain (and any opener chain) looking for the LMS API object.
  function findAPI(propName) {
    var seen = [], w = window, g = 0;
    while (w && g++ < 12) {
      seen.push(w);
      if (w.parent === w) { break; }
      try { w = w.parent; } catch (e) { break; }
    }
    try {
      var o = window.opener, g2 = 0;
      while (o && g2++ < 6) {
        seen.push(o);
        if (o.parent === o) { break; }
        try { o = o.parent; } catch (e) { break; }
      }
    } catch (e) {}
    for (var i = 0; i < seen.length; i++) {
      try { if (seen[i][propName]) { return seen[i][propName]; } } catch (e) {}
    }
    return null;
  }

  function scormLearner() {
    // SCORM 2004 first
    var api4 = findAPI("API_1484_11");
    if (api4 && typeof api4.GetValue === "function") {
      try {
        var id4 = api4.GetValue("cmi.learner_id");
        var nm4 = api4.GetValue("cmi.learner_name");
        if (id4 || nm4) { return { id: id4, name: nm4, ver: "2004" }; }
      } catch (e) { log("SCORM 2004 read failed.", e); }
    }
    // SCORM 1.2
    var api2 = findAPI("API");
    if (api2 && typeof api2.LMSGetValue === "function") {
      try {
        var id2 = api2.LMSGetValue("cmi.core.student_id");
        var nm2 = api2.LMSGetValue("cmi.core.student_name");
        if (id2 || nm2) { return { id: id2, name: nm2, ver: "1.2" }; }
      } catch (e) { log("SCORM 1.2 read failed.", e); }
    }
    return null;
  }

  // "Doe, Jane" -> "Jane Doe"
  function tidyName(n) {
    if (!n) { return null; }
    n = String(n).trim();
    if (n.indexOf(",") > -1) {
      var p = n.split(",");
      n = ((p[1] || "").trim() + " " + (p[0] || "").trim()).trim();
    }
    n = n.replace(/\s+/g, " ").trim();
    return n || null;
  }

  function lmsHomePage() {
    try {
      var ao = window.location.ancestorOrigins;
      if (ao && ao.length) {
        var o = ao[ao.length - 1];
        if (o && o !== "null") { return o; }
      }
    } catch (e) {}
    try {
      var to = window.top.location.origin;
      if (to && to !== "null") { return to; }
    } catch (e) {}
    return (CONFIG.fallbackActor.account && CONFIG.fallbackActor.account.homePage) || "http://example.org";
  }

  function actorFromScorm(s) {
    var name = tidyName(s.name) || CONFIG.fallbackActor.name;
    var id = String(s.id || "").trim();
    if (!id) { return { name: name, account: { homePage: lmsHomePage(), name: name } }; }
    // If the LMS uses an email address as the learner ID, use a proper mbox.
    if (/^[^@\s]+@[^@\s]+\.[^@\s]+$/.test(id)) {
      return { name: name, mbox: "mailto:" + id };
    }
    return { name: name, account: { homePage: lmsHomePage(), name: id } };
  }

  // ---------- xAPI launch (for xAPI/Tin Can publishes) ----------
  function launchActor() {
    var urls = frameUrls();
    for (var i = 0; i < urls.length; i++) {
      var a = getParam(urls[i], "actor");
      if (a) {
        try { return JSON.parse(a); }
        catch (e) { log("Could not parse launch actor.", e); }
      }
    }
    return null;
  }

  // ---------- send ----------
  function send(actor) {
    var stmt = {
      actor: actor,
      verb: CONFIG.verb,
      object: CONFIG.object,
      timestamp: new Date().toISOString()
    };
    if (CONFIG.result) { stmt.result = CONFIG.result; }

    // Proxy mode: post straight to the worker, which holds the credential.
    // Direct mode: append /statements and send the Authorization header.
    var url = CONFIG.proxyMode
      ? CONFIG.endpoint
      : CONFIG.endpoint.replace(/\/+$/, "") + "/statements";
    var xhr = new XMLHttpRequest();
    xhr.open("POST", url, true);
    xhr.setRequestHeader("Content-Type", "application/json");
    xhr.setRequestHeader("X-Experience-API-Version", CONFIG.version);
    if (!CONFIG.proxyMode && CONFIG.auth) {
      xhr.setRequestHeader("Authorization", CONFIG.auth);
    }
    xhr.onreadystatechange = function () {
      if (xhr.readyState !== 4) { return; }
      if (xhr.status >= 200 && xhr.status < 300) {
        log("Statement accepted (" + xhr.status + ").", stmt);
        if (CONFIG.onceKey) {
          try { localStorage.setItem(CONFIG.onceKey, String(Date.now())); } catch (e) {}
        }
      } else {
        log("Statement failed (" + xhr.status + ").", xhr.responseText);
      }
    };
    try { xhr.send(JSON.stringify(stmt)); }
    catch (e) { log("Send error.", e); }
  }

  // The SCORM API may not be wired up the instant this block loads,
  // so retry briefly before giving up and using the fallback actor.
  var attempts = 0, MAX_ATTEMPTS = 8, RETRY_MS = 300;
  function promptActor() {
    var nm = (window.prompt("Please enter your name:") || "").trim() || CONFIG.fallbackActor.name;
    var em = (window.prompt("Please enter your email:") || "").trim();
    if (em) { return { name: nm, mbox: "mailto:" + em }; }
    return { name: nm, account: { homePage: lmsHomePage(), name: nm } };
  }

  function resolveAndSend() {
    if (CONFIG.actorSource === "static") { send(CONFIG.fallbackActor); return; }
    if (CONFIG.actorSource === "prompt") { send(promptActor()); return; }

    var s = scormLearner();
    if (s) {
      var a = actorFromScorm(s);
      log("Learner identified via SCORM " + s.ver + ".", a);
      send(a);
      return;
    }
    var la = launchActor();
    if (la) {
      log("Learner identified via xAPI launch.", la);
      send(la);
      return;
    }
    if (attempts++ < MAX_ATTEMPTS) {
      setTimeout(resolveAndSend, RETRY_MS);
      return;
    }
    log("No learner found after " + MAX_ATTEMPTS + " attempts - using fallback actor.");
    send(CONFIG.fallbackActor);
  }

  if (document.readyState === "loading") {
    document.addEventListener("DOMContentLoaded", resolveAndSend);
  } else {
    resolveAndSend();
  }
})();
</script>
